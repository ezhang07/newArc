# Runtime dependencies — what the FastAPI server needs in production.
# Render installs THIS file. Keep it minimal so deploy builds stay fast.
#
# Dev/notebook tooling (jupyter, scikit-learn, requests) lives in
# requirements-dev.txt so it never bloats the production build.

fastapi         # web framework for the /recommend + /anime API
uvicorn         # ASGI server that runs the FastAPI app
pydantic        # request/response validation (also pulled in by fastapi)
pandas          # load the catalog parquet + wrangle columns
numpy           # vector math for taste vectors + cosine similarity
pyarrow         # parquet reader (preserves list columns like genres/themes)
